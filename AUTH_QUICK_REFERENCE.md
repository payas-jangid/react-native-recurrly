# Clerk Auth - Quick Reference\n\n## Common Usage Patterns\n\n### Get Current User\n`tsx\nimport { useUser } from '@clerk/expo';\n\nconst { user } = useUser();\nconst email = user?.emailAddresses[0]?.emailAddress;\nconst name = user?.firstName;\n`\n\n### Check if Signed In\n`tsx\nimport { useAuth } from '@clerk/expo';\n\nconst { isSignedIn, isLoaded } = useAuth();\n\nif (!isLoaded) return <ActivityIndicator />;\nif (!isSignedIn) return <SignInScreen />;\nreturn <HomeScreen />;\n`\n\n### Sign Out\n`tsx\nimport { useClerk } from '@clerk/expo';\n\nconst { signOut } = useClerk();\n\n<Pressable onPress={() => signOut()}>\n  <Text>Sign Out</Text>\n</Pressable>\n`\n\n### Display User Email\n`tsx\nimport { useDisplayEmail } from '@/lib/auth';\n\nconst email = useDisplayEmail();\n<Text>Signed in as {email}</Text>\n`\n\n### Get User Initials for Avatar\n`tsx\nimport { getUserInitials } from '@/lib/auth';\n\nconst initials = getUserInitials(email);\n<Text>{initials}</Text>\n`\n\n### Mask Email for Privacy\n`tsx\nimport { maskEmail } from '@/lib/auth';\n\nconst masked = maskEmail('john.doe@example.com');\n// Output: j***o@example.com\n`\n\n### Validate Email\n`tsx\nimport { isValidEmail } from '@/lib/auth';\n\nconst valid = isValidEmail('user@example.com');\n`\n\n### Validate Password\n`tsx\nimport { isValidPassword, isStrongPassword } from '@/lib/auth';\n\nconst meetsMinimum = isValidPassword(password); // 8+ chars\nconst isStrong = isStrongPassword(password); // uppercase, lowercase, number\n`\n\n### Format Auth Errors\n`tsx\nimport { formatAuthError } from '@/lib/auth';\n\ntry {\n  await signIn.password({ emailAddress, password });\n} catch (error) {\n  const message = formatAuthError(error);\n  setError(message);\n}\n`\n\n### Check Specific Error Code\n`tsx\nimport { isClerkErrorCode, CLERK_ERROR_CODES } from '@/lib/auth';\n\nif (isClerkErrorCode(error, CLERK_ERROR_CODES.USER_NOT_FOUND)) {\n  // Handle user not found\n}\n`\n\n## Auth Utilities\n\n### Available in `lib/auth.ts`\n\n| Function | Purpose |\n|----------|----------|\n| `isValidEmail()` | Check email format |\n| `isValidPassword()` | Check password length (8+) |\n| `isStrongPassword()` | Check password strength |\n| `useDisplayEmail()` | Hook to get formatted user email |\n| `useIsAuthenticated()` | Hook to check auth status |\n| `useAuthState()` | Hook to get full auth state |\n| `useSignOutHandler()` | Hook for safe sign out |\n| `formatAuthError()` | Convert error to display text |\n| `isClerkErrorCode()` | Check specific error code |\n| `getUserInitials()` | Get avatar initials |\n| `maskEmail()` | Mask email for privacy |\n| `debounce()` | Debounce function calls |\n\n## Styling Classes\n\nAll auth screens use these NativeWind classes from `global.css`:\n\n| Class | Purpose |\n|-------|----------|\n| `.auth-safe-area` | Full screen container |\n| `.auth-content` | Content wrapper |\n| `.auth-brand-block` | Logo and title area |\n| `.auth-card` | Form card wrapper |\n| `.auth-form` | Form container |\n| `.auth-field` | Form field wrapper |\n| `.auth-label` | Input label |\n| `.auth-input` | Text input |\n| `.auth-input-error` | Error state input |\n| `.auth-error` | Error message text |\n| `.auth-button` | Primary button |\n| `.auth-button-disabled` | Disabled button state |\n| `.auth-button-text` | Button text |\n| `.auth-secondary-button` | Secondary button |\n| `.auth-secondary-button-text` | Secondary button text |\n| `.auth-link-row` | Sign in/up link row |\n| `.auth-link` | Navigation link |\n\n## Environment Variables\n\n`env\n# Required - Get from Clerk Dashboard → API Keys\nEXPO_PUBLIC_CLERK_PUBLISHABLE_KEY=pk_test_your_key\n\n# Optional - For additional Clerk features\nEXPO_PUBLIC_CLERK_SIGN_IN_URL=/sign-in\nEXPO_PUBLIC_CLERK_SIGN_UP_URL=/sign-up\n`\n\n## Type Definitions\n\nAvailable in `type.auth.d.ts`:\n\n`tsx\ntype AuthState = {\n  isLoaded: boolean;\n  isSignedIn: boolean;\n  user: UserResource | null;\n  isAuthenticating: boolean;\n  isAuthenticated: boolean;\n};\n\ntype SignUpFormState = {\n  emailAddress: string;\n  password: string;\n  code: string;\n  isVerifying: boolean;\n  errors: Record<string, string>;\n  statusMessage: string;\n};\n\ntype UserProfile = {\n  id: string;\n  email: string;\n  firstName?: string | null;\n  lastName?: string | null;\n  fullName?: string;\n  imageUrl?: string;\n  createdAt: Date;\n};\n`\n\n## Common Error Codes\n\nFrom `lib/auth.ts` `CLERK_ERROR_CODES`:\n\n| Code | Meaning |\n|------|----------|\n| `USER_NOT_FOUND` | Email not registered |\n| `INVALID_PASSWORD` | Wrong password |\n| `PASSWORD_TOO_SHORT` | Password < 8 characters |\n| `EMAIL_EXISTS` | Email already in use |\n| `INVALID_EMAIL` | Email format invalid |\n| `TOO_MANY_ATTEMPTS` | Rate limited |\n| `SESSION_EXPIRED` | Session timed out |\n\n## Route Structure\n\n`\napp/\n├── _layout.tsx                    (Root + ClerkProvider)\n├── (auth)/\n│   ├── _layout.tsx               (Auth guard)\n│   ├── sign-in.tsx               (Email/password sign-in)\n│   └── sign-up.tsx               (Email/password sign-up)\n├── (tabs)/\n│   ├── _layout.tsx               (Signed-in guard)\n│   ├── index.tsx                 (Home)\n│   ├── insights.tsx              (Insights)\n│   ├── subscriptions.tsx          (Subscriptions)\n│   └── settings.tsx              (Settings + Sign out)\n└── onboarding.tsx                (Optional welcome screen)\n`\n\n## Debug Tips\n\n### Check Auth State\n`tsx\nconst { isLoaded, isSignedIn } = useAuth();\nconsole.log('Auth loaded:', isLoaded);\nconsole.log('Signed in:', isSignedIn);\n`\n\n### View Current User\n`tsx\nconst { user } = useUser();\nconsole.log('User:', user);\nconsole.log('Email:', user?.emailAddresses[0]?.emailAddress);\n`\n\n### Check Sign-In Status\n`tsx\nconst { signIn, status } = useSignIn();\nconsole.log('Sign-in status:', status);\n// Possible values: 'needs_identifier', 'needs_factor_one', 'complete', etc.\n`\n\n### View Error Details\n`tsx\nconst { errors } = useSignIn();\nconsole.log('Errors:', JSON.stringify(errors, null, 2));\n`\n\n## Performance Tips\n\n1. **Memoize Auth State**: Use `useMemo()` in custom hooks to prevent re-renders\n2. **Lazy Load Protected Screens**: Only render tabs after `isLoaded` is true\n3. **Debounce Validation**: Use `debounce()` for email/password validation\n4. **Cache User Data**: Store user profile in local state, not re-fetching\n\n## Security Checklist\n\n- ✅ Tokens stored in `expo-secure-store` (encrypted)\n- ✅ HTTPS only for all Clerk API calls\n- ✅ Password fields use `secureTextEntry`\n- ✅ No sensitive data in console logs\n- ✅ Session validation on app launch\n- ✅ Automatic token refresh\n\n## Support Resources\n\n- [Clerk Expo
